\begin{document}
30 minutes maximum.  No aids (book, calculator, etc.) are permitted.
Show all work and use proper notation for full credit.  Answers should
be in reasonably-simplified form.  25 points possible.

\begin{enumerate}


  \item Consider the sequence defined by $a_1 = 2$ and $\ds a_n =
    -\frac{1}{2} a_{n-1}$.
    \begin{enumerate}
    \item (2 points) Write out the first 5 terms of the sequence. Simplify each
      term.
      \begin{solution}
        \[
          \boxed{a_1 = 2,\quad a_2 = -1,\quad a_3 = \tfrac12,\quad a_4 = -\tfrac14,\quad a_5 = \tfrac18}
        \]
      \end{solution}
      \vfill

    \item (4 points) Find a formula (closed form) for the $n$th term $a_n$ of
      the sequence.
      \begin{solution}
        $\ds\boxed{a_n = 2\left(-\frac12\right)^{n-1}}$
      \end{solution}
      \vfill

    \item (3 points) Is the sequence bounded? Justify your answer. If it is
      bounded, clearly state the bounds.
      \begin{solution}
        Yes. $|a_n| = 2\left(\frac12\right)^{n-1}$ shrinks as $n$ grows, so the largest
        positive term is $a_1 = 2$ and the most negative term is $a_2 = -1$:\quad
        $\boxed{-1 \le a_n \le 2 \ \text{ for all } n}$
      \end{solution}
      \vfill

    \item (3 points) Is the sequence monotone? Justify your answer.
      \begin{solution}
        The signs of $a_n$ alternate for every $n$.\quad
        $\boxed{\text{Not monotone}}$
      \end{solution}
      \vfill

    \item (3 points) What (if anything) can you say about the sequence
      based on your answers to
      parts (c) and (d)?
      \begin{solution}
        The Monotone Convergence Theorem needs a sequence that is both bounded and monotone.
        This one is not monotone, so the theorem does not apply.
        \[
          \boxed{\text{Nothing: (c) and (d) do not determine whether } \{a_n\} \text{ converges.}}
        \]
      \end{solution}
      \vfill
      
    \end{enumerate}

    \newpage

  \item Compute the following integrals or show that they
    diverge. Show all work and use proper notation.
    \begin{enumerate}
    \item (5 points) $\ds \int_1^{\infty} \frac{\ln(x)}{x} \, dx$
      \begin{solution}
        \[
          \int_1^{\infty} \frac{\ln x}{x}\,dx = \lim_{t\to\infty} \int_1^t \frac{\ln x}{x}\,dx.
        \]
        With $u = \ln x$, $du = \frac{dx}{x}$: $\ds\int \frac{\ln x}{x}\,dx = \frac{(\ln x)^2}{2} + C$, so
        \[
          \lim_{t\to\infty} \left[\frac{(\ln x)^2}{2}\right]_1^t
          = \lim_{t\to\infty} \frac{(\ln t)^2}{2} = \infty.
        \]
        \[
          \boxed{\int_1^{\infty} \frac{\ln x}{x}\,dx \ \text{ diverges}}
        \]
      \end{solution}
      \vfill
      
    \item (5 points) $\ds \int_0^9 \frac{dx}{\sqrt{x}}$
      \begin{solution}
        \[
          \int_0^9 \frac{dx}{\sqrt{x}} = \lim_{t\to 0^+} \int_t^9 x^{-1/2}\,dx
          = \lim_{t\to 0^+} \Big[2\sqrt{x}\Big]_t^9
          = \lim_{t\to 0^+}\big(6 - 2\sqrt{t}\big) = \boxed{6}
        \]
      \end{solution}
      \vfill
      

    \item BONUS (2 points) Compute the following integral. $\ds \int_0^3 \frac{1}{x-1}
    \, dx$
      \begin{solution}
        The integrand is undefined at $x = 1$, inside $[0,3]$:
        \[
          \int_0^3 \frac{dx}{x-1} = \lim_{s\to 1^-} \int_0^s \frac{dx}{x-1} + \lim_{t\to 1^+} \int_t^3 \frac{dx}{x-1}.
        \]
        \[
          \lim_{t\to 1^+} \int_t^3 \frac{dx}{x-1} = \lim_{t\to 1^+} \Big[\ln|x-1|\Big]_t^3
          = \lim_{t\to 1^+}\big(\ln 2 - \ln(t-1)\big) = \infty.
        \]
        Since one piece diverges,
        \[
          \boxed{\int_0^3 \frac{dx}{x-1} \ \text{ diverges}}
        \]
      \end{solution}
    \vfill

  \end{enumerate}

\end{enumerate}
