\medskip \noindent \textbf{2.} \quad $\ds \sum_{n=1}^\infty \frac{\ln(n)}{n}$

\vspace{22mm}
\medskip \noindent \textbf{3.} \quad $\ds \sum_{n=1}^\infty \frac{1}{n^3-13}$
